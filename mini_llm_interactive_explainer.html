# Split into heads: (T, H, d_head) Q = (x @ Wq).reshape(T, n_heads, d_head).transpose(1,0,2) # H,T,dh K = (x @ Wk).reshape(T, n_heads, d_head).transpose(1,0,2) V = (x @ Wv).reshape(T, n_heads, d_head).transpose(1,0,2) # Scaled dot-product attention + causal mask scores = (Q @ K.transpose(0,2,1)) / sqrt(d_head) + causal_mask attn = softmax(scores, axis=-1) # (H, T, T) ctx = attn @ V # (H, T, dh) out = ctx.transpose(1,0,2).reshape(T, d_model) @ Wo # Residual + LayerNorm x = layer_norm(x + out, gamma, beta)
